\subsection*{Glossary}
\begin{table}[H]
	\renewcommand{\arraystretch}{1.5}
	\begin{tabular}{|l|p{11cm}|}
		\hline
		\textbf{Term}                & \textbf{Definition} \\ \hline
		Automation bias              & The tendency of human operators to over-rely on automated system outputs, even when they have the authority to override them \citep{parasuraman2010complacency}. \\ \hline
		Contrastive explanation      & An explanation that answers ``why X rather than Y'' by comparing a decision against an alternative outcome \citep{miller2018explanationartificialintelligenceinsights}. \\ \hline
		Ensemble architecture        & A machine learning system composed of multiple specialised models whose outputs are combined to produce a unified prediction. \\ \hline
		Glass Box                    & A compliance monitoring approach that defines observable norms at a system's input/output boundary, enabling oversight without access to internal model logic \citep{info12090385}. \\ \hline
		Indirect stakeholder         & A party affected by a system's outputs without directly interacting with it \citep{friedman2013vsd}. \\ \hline
		Positive identification      & The confirmed determination that a target is a legitimate military objective, as required by IHL's distinction principle \citep{ootlaws}. \\ \hline
		Post-hoc explainability      & Explanation methods applied after a model has made a prediction, rather than built into the model's architecture \citep{arrieta2019explainableartificialintelligencexai}. \\ \hline
	\end{tabular}
\end{table}
